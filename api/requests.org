* App API                                                         :verb:
Template http://localhost:8000/app
# Access-Token: Bearer TOKEN

** Health check                                                       :verb:
GET /health
# Returns: 200 OK with {"status": "healthy"} when healthy
# Returns: 503 Service Unavailable with {"status": "unhealthy"} when unhealthy

** Sign up (create org)
POST /auth/signup
Content-Type: application/json
# Returns: 201 Created with {access_token, user, membership}
# Errors: 409 conflict (email taken), 422 validation_error (weak password,
# is_admin/role/org_id in body, or not exactly one of org_name/invite_token)

{
  "email": "ada@example.com",
  "password": "longenough1",
  "name": "Ada Lovelace",
  "org_name": "Analytical Engines"
}

** Sign up (accept invite)
POST /auth/signup
Content-Type: application/json
# Returns: 201 Created with {access_token, user, membership}
# Errors: 404 invite_invalid, 409 conflict (email taken), 422 validation_error

{
  "email": "grace@example.com",
  "password": "longenough1",
  "invite_token": "TOKEN"
}

** Log in
POST /auth/login
Content-Type: application/json
# Returns: 200 OK with {access_token}
# Errors: 401 unauthorized (generic — wrong password, unknown email or
# deactivated user), 422 validation_error

{
  "email": "ada@example.com",
  "password": "longenough1"
}

** Log out
POST /auth/logout
# Returns: 200 OK with {"message": "Successfully logged out"}

** Get current user + memberships
GET /me
# Returns: 200 OK with {user, memberships: [{org, role}]}
# Requires: Authentication

** Create another org (existing user becomes owner)
POST /orgs
Content-Type: application/json
# Returns: 201 Created with the org
# Requires: Authentication

{
  "name": "Second Org"
}

** Preview invite
GET /invites/TOKEN
# Returns: 200 OK with {org_name, role}
# Returns: 404 invite_invalid for invalid/expired/used/revoked invites

** Accept invite
POST /invites/TOKEN/accept
# Returns: 201 Created with {membership: {org_id, role}}
# Errors: 404 invite_invalid, 409 conflict (already a member)
# Requires: Authentication

** Get org (org-scoped)
GET /orgs/ORG_ID
# Returns: 200 OK with the org
# Requires: a member of the org; 404 not_found for non-members AND unknown
# orgs (identical body, org existence is never disclosed)

** Rename org (org-scoped)
PATCH /orgs/ORG_ID
Content-Type: application/json
# Returns: 200 OK with the renamed org (slug is immutable)
# Requires: admin or owner; 403 forbidden otherwise
# Errors: 422 validation_error (extra/forbidden fields like role, org_id)

{
  "name": "Northwind Labs 2"
}

** List members (org-scoped)
GET /orgs/ORG_ID/members?limit=50&offset=0
# Returns: 200 OK with {items, total, limit, offset}; items are
# {membership_id, user: {id, name, email}, role, created_at}
# Requires: admin or owner

** Change member role (org-scoped)
PATCH /orgs/ORG_ID/members/MEMBERSHIP_ID
Content-Type: application/json
# Returns: 200 OK with the updated membership
# Errors: 403 forbidden (admin touching an owner), 409 last_owner (last owner
# can never be demoted), 404 not_found (member not in this org)
# Requires: admin or owner

{
  "role": "admin"
}

** Remove member (org-scoped)
DELETE /orgs/ORG_ID/members/MEMBERSHIP_ID
# Returns: 204 No Content (team_members rows and plans cascade; team lead links
# are cleared)
# Errors: 403 forbidden (admin touching an owner), 409 last_owner,
# 404 not_found (member not in this org)
# Requires: admin or owner

** Create invite (org-scoped)
POST /orgs/ORG_ID/invites
Content-Type: application/json
# Returns: 201 Created with {id, token, role, expires_at, url}
# Errors: 422 validation_error (role must be admin|member — never owner),
# 403 forbidden
# Requires: admin or owner

{
  "role": "member",
  "expires_in_days": 7
}

** List valid invites (org-scoped)
GET /orgs/ORG_ID/invites?limit=50&offset=0
# Returns: 200 OK with {items, total, limit, offset}; only invites that still
# work (unused, unrevoked, unexpired) are listed
# Requires: admin or owner

** Revoke invite (org-scoped)
DELETE /orgs/ORG_ID/invites/INVITE_ID
# Returns: 204 No Content (idempotent)
# Returns: 404 not_found (invite not in this org)
# Requires: admin or owner

* Internal API                                                            :verb:
Template http://localhost:8000/internal
# Access-Token: Bearer TOKEN

** Health check                                                       :verb:
GET /health
# Returns: 200 OK with {"status": "healthy"} when healthy
# Returns: 503 Service Unavailable with {"status": "unhealthy"} when unhealthy

** Authenticate
POST /auth
Content-Type: application/json
# Returns: 200 OK with access token
# Note: only users with is_admin may authenticate here
# Errors: 401 unauthorized (generic)

{
  "email": "admin@example.com",
  "password": "adminpass1"
}

** Logout
POST /logout
# Returns: 200 OK with {"message": "Successfully logged out"}

** Get current internal user
GET /users/self
# Returns: 200 OK with internal user details
# Requires: Internal authentication (otherwise 401; 403 staff_only if the
# token's user is not is_admin)

** List users
GET /users?limit=50&offset=0
# Returns: 200 OK with {items, total, limit, offset}
# Query: limit (1-100, default 50), offset (>=0, default 0)
# Requires: Internal authentication

** Create user
POST /users
Content-Type: application/json
# Returns: 201 Created with the new user
# Returns: 409 Conflict if the email already exists
# Requires: Internal authentication

{
  "first_name": "Jane",
  "last_name": "Doe",
  "email": "jane@example.com",
  "password": "secret123",
  "is_admin": false,
  "is_active": true
}

** Update user (partial)
PATCH /users/{{(verb-var user-id)}}
Content-Type: application/json
# Returns: 200 OK with the updated user
# Note: only sent fields change; password is ignored here (use change-password)
# Returns: 404 if not found, 409 on email conflict
# Requires: Internal authentication

{
  "last_name": "Smith",
  "is_admin": true
}

** Change user password
POST /users/{{(verb-var user-id)}}/change-password
Content-Type: application/json
# Returns: 204 No Content
# Requires: Internal authentication

{
  "password": "brandnew1"
}

** Delete user
DELETE /users/{{(verb-var user-id)}}
# Returns: 204 No Content (row hard-deleted)
# Returns: 404 if not found
# Requires: Internal authentication
